\noindent\textbf{Listing A.1 — RRF fusion (\texttt{src/retrieval/hybrid\_retriever.py})}

\begin{lstlisting}[language=Python]
def reciprocal_rank_fusion(
    self,
    bm25_results: List[Dict],
    dense_results: List[Dict],
    k: int = 60
) -> List[Dict]:
    """Merge BM25 and dense ranked lists using Reciprocal Rank Fusion."""
    scores: Dict[str, float] = {}
    doc_map: Dict[str, Dict] = {}

    for rank, doc in enumerate(bm25_results):
        doc_id = doc["id"]
        scores[doc_id] = scores.get(doc_id, 0) + 1.0 / (k + rank + 1)
        doc_map[doc_id] = doc

    for rank, doc in enumerate(dense_results):
        doc_id = doc["id"]
        scores[doc_id] = scores.get(doc_id, 0) + 1.0 / (k + rank + 1)
        doc_map[doc_id] = doc

    ranked = sorted(scores.items(), key=lambda x: x[1], reverse=True)
    return [doc_map[doc_id] for doc_id, _ in ranked]
\end{lstlisting}

\noindent\textbf{Listing A.2 --- Five-signal confidence scorer} (\path{src/xai/multi_signal_confidence.py})

\begin{lstlisting}[language=Python]
def compute_confidence(
    self,
    query: str,
    answer: str,
    retrieved_docs: List[Dict],
    generation_scores: Optional[List[float]] = None
) -> Dict[str, float]:
    """Compute five confidence signals and return combined score."""
    retrieval_conf = self._retrieval_confidence(retrieved_docs)
    generation_conf = self._generation_confidence(generation_scores)
    consistency   = self._self_consistency(query, answer)
    source_agree  = self._source_agreement(retrieved_docs)
    entity_cov    = self._entity_coverage(query, answer)

    weights = [0.25, 0.25, 0.20, 0.20, 0.10]
    signals = [retrieval_conf, generation_conf,
               consistency, source_agree, entity_cov]
    raw = sum(w * s for w, s in zip(weights, signals))
    calibrated = self.platt_scale(raw)

    return {
        "combined": calibrated,
        "retrieval": retrieval_conf,
        "generation": generation_conf,
        "consistency": consistency,
        "source_agreement": source_agree,
        "entity_coverage": entity_cov,
    }
\end{lstlisting}

# Separator hierarchy: paragraph -> line -> sentence -> clause -> word
_SEPARATORS = ["\n\n", "\n", ". ", "; ", ", ", " ", ""]

def _split(self, text: str, separators: List[str],
           chunk_size: int) -> List[str]:
    """Recursively split text using the separator hierarchy."""
    if not text:
        return []
    sep = ""
    remaining_seps = separators[:]
    for candidate in separators:
        if candidate == "" or candidate in text:
            sep = candidate
            remaining_seps = separators[separators.index(candidate) + 1:]
            break
    if sep:
        pieces = re.split(re.escape(sep), text)
    else:
        return [text[i: i + chunk_size]
                for i in range(0, len(text), chunk_size)]

    good: List[str] = []
    for piece in pieces:
        piece = piece.strip()
        if not piece:
            continue
        if len(piece) <= chunk_size:
            good.append(piece)
        else:
            good.extend(self._split(piece, remaining_seps, chunk_size))

    merged = self._merge(good, chunk_size)
    return self._add_overlap(merged) if self._chunk_overlap > 0 else merged

\noindent\textbf{Listing A.4 — Safety guardrail pre-check (\texttt{src/safety/guardrails.py})}

\begin{lstlisting}[language=Python]
EMERGENCY_KEYWORDS = [
    "suicide", "suicidal", "kill myself", "end my life",
    "heart attack", "can't breathe", "chest pain",
    "stroke symptoms", "severe bleeding", "unconscious",
    "overdose", "poisoning", "anaphylaxis",
    "severe allergic reaction", "choking",
]

DANGEROUS_PATTERNS = [
    r"you\s+should\s+take\s+\d+\s*(mg|ml|pills|tablets)",
    r"you\s+(have|definitely\s+have)\s+\w+\s+(disease|disorder|syndrome|cancer)",
    r"stop\s+taking\s+your\s+(medication|medicine|prescription)",
    r"you\s+don'?t\s+need\s+to\s+see\s+a\s+doctor",
    r"instead\s+of\s+(seeing\s+a\s+doctor|going\s+to|medical\s+care)",
]

def check_safety(self, query: str) -> Optional[SafetyResult]:
    """Return a SafetyResult if query matches an emergency keyword."""
    text_lower = query.lower()
    for keyword in self.EMERGENCY_KEYWORDS:
        if keyword in text_lower:
            return SafetyResult(
                level=SafetyLevel.EMERGENCY,
                redirect_message=self.EMERGENCY_MESSAGE,
                flags=["emergency_detected"],
            )
    return None
\end{lstlisting}
